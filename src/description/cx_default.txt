def cx_operator(input_ind1, input_ind2, env):
    try:
        ind1 = copy.deepcopy(input_ind1)
        ind2 = copy.deepcopy(input_ind2)

        num_routes = min(len(ind1), len(ind2))
        if num_routes == 0:
            if len(ind1) < len(input_ind1):
                ind1.extend(copy.deepcopy(input_ind1[len(ind1):]))
            if len(ind2) < len(input_ind2):
                ind2.extend(copy.deepcopy(input_ind2[len(ind2):]))
            return ind1, ind2

        tourist_map = env.tourist_map
        distance_map = env.distance_map
        PI_map = env.PI_map
        NI_map = env.NI_map

        def get_edge_crossover_score(u, v):
            u, v = int(u), int(v)
            key = (u, v)
            if not env.G.has_edge(u, v):
                return float('inf')
            length = distance_map.get(key, 1.0)
            pi = PI_map.get(key, 0.0)
            ni = NI_map.get(key, 0.0)
            visual_score = (pi - ni) / (length + 1e-6)
            demand_u = tourist_map.get(u, 0)
            demand_v = tourist_map.get(v, 0)
            node_demand_bonus = demand_u + demand_v
            edge_score = -visual_score
            if node_demand_bonus == 0:
                edge_score += 2.5  # Increased penalty for zero-demand edges
            return edge_score

        def select_cut_points(route):
            if len(route) < 4:
                return [len(route) // 2]
            scores = []
            for i in range(1, len(route) - 1):
                u, v = route[i-1], route[i]
                score = get_edge_crossover_score(u, v)
                scores.append((score, i))
            sorted_scores = sorted(scores, key=lambda x: x[0], reverse=True)
            cuts = [idx for _, idx in sorted_scores[:2]]
            return sorted(cuts) if cuts else [len(route) // 2]

        def repair_path_segment(start_node, end_node):
            if start_node == end_node:
                return []
            try:
                def weight_func(u_, v_, d):
                    u_int, v_int = int(u_), int(v_)
                    length = d.get('length', 1.0)
                    pi = d.get('PI', 0.0)
                    ni = d.get('NI', 0.0)
                    visual_factor = (pi - ni) / (length + 1e-6)
                    demand_u = tourist_map.get(u_int, 0)
                    demand_v = tourist_map.get(v_int, 0)
                    demand_score = demand_u + demand_v
                    flow_val = env.flow.get(u_int, {}).get(v_int, 0.0)
                    return length / (1e-4 + visual_factor + 0.1 * demand_score + 1e-6) + flow_val * 0.1

                path = nx.shortest_path(env.G, source=start_node, target=end_node, weight=weight_func, method='dijkstra')
                return path[1:]
            except (nx.NetworkXNoPath, nx.NodeNotFound, KeyError):
                try:
                    path = nx.shortest_path(env.G, source=start_node, target=end_node, weight='length', method='dijkstra')
                    return path[1:]
                except:
                    return [end_node]

        def repair_full_route(route):
            if len(route) < 2:
                return route
            repaired = [int(route[0])]
            visited = {repaired[0]}
            max_nodes = 1000
            for i in range(1, len(route)):
                if len(repaired) >= max_nodes:
                    break
                next_node = int(route[i])
                if next_node == repaired[-1]:
                    continue
                if next_node in visited and i != len(route) - 1:
                    continue
                if not env.G.has_edge(repaired[-1], next_node):
                    segment = repair_path_segment(repaired[-1], next_node)
                    for node in segment:
                        node_int = int(node)
                        if node_int not in visited or node_int == next_node:
                            if len(repaired) >= max_nodes:
                                break
                            repaired.append(node_int)
                            visited.add(node_int)
                else:
                    if next_node not in visited or i == len(route) - 1:
                        repaired.append(next_node)
                        visited.add(next_node)
            return repaired

        for i in range(num_routes):
            route1 = [int(n) for n in ind1[i]]
            route2 = [int(n) for n in ind2[i]]

            if len(route1) < 3 or len(route2) < 3:
                continue

            start1, end1 = route1[0], route1[-1]
            start2, end2 = route2[0], route2[-1]

            cuts1 = select_cut_points(route1)
            cuts2 = select_cut_points(route2)

            cxpoint1 = min(cuts1[0], len(route1) - 2) if cuts1 else len(route1) // 2
            cxpoint2 = min(cuts2[0], len(route2) - 2) if cuts2 else len(route2) // 2

            head1 = route1[:cxpoint1]
            tail1 = route1[cxpoint1:]
            head2 = route2[:cxpoint2]
            tail2 = route2[cxpoint2:]

            new_route1 = head1 + tail2
            new_route2 = head2 + tail1

            try:
                new_route1 = repair_full_route(new_route1)
                new_route2 = repair_full_route(new_route2)
            except Exception:
                continue

            if not new_route1 or new_route1[0] != start1:
                new_route1.insert(0, start1)
            if new_route1[-1] != end1:
                new_route1.append(end1)

            if not new_route2 or new_route2[0] != start2:
                new_route2.insert(0, start2)
            if new_route2[-1] != end2:
                new_route2.append(end2)

            ind1[i] = new_route1
            ind2[i] = new_route2

        if len(ind1) < len(input_ind1):
            ind1.extend(copy.deepcopy(input_ind1[len(ind1):]))
        elif len(ind1) > len(input_ind1):
            ind1 = ind1[:len(input_ind1)]

        if len(ind2) < len(input_ind2):
            ind2.extend(copy.deepcopy(input_ind2[len(ind2):]))
        elif len(ind2) > len(input_ind2):
            ind2 = ind2[:len(input_ind2)]

        return ind1, ind2

    except Exception as e:
        return copy.deepcopy(input_ind1), copy.deepcopy(input_ind2)
